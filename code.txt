import os
import tkinter as tk
from tkinter import filedialog, messagebox
from getpass import getpass
from cryptography.hazmat.primitives.kdf.pbkdf2 import PBKDF2HMAC
from cryptography.hazmat.primitives import hashes
from cryptography.hazmat.primitives.ciphers import Cipher, algorithms, modes
from cryptography.hazmat.backends import default_backend

BLOCK_SIZE = 16
SALT_SIZE = 16
KEY_SIZE = 32
ITERATIONS = 100_000

def derive_key(password: str, salt: bytes) -> bytes:
    kdf = PBKDF2HMAC(
        algorithm=hashes.SHA256(),
        length=KEY_SIZE,
        salt=salt,
        iterations=ITERATIONS,
        backend=default_backend()
    )
    return kdf.derive(password.encode())

def encrypt_file(input_path, password):
    salt = os.urandom(SALT_SIZE)
    key = derive_key(password, salt)
    iv = os.urandom(BLOCK_SIZE)

    with open(input_path, 'rb') as f:
        plaintext = f.read()

    padding_length = BLOCK_SIZE - len(plaintext) % BLOCK_SIZE
    plaintext += bytes([padding_length]) * padding_length

    cipher = Cipher(algorithms.AES(key), modes.CBC(iv), backend=default_backend())
    encryptor = cipher.encryptor()
    ciphertext = encryptor.update(plaintext) + encryptor.finalize()

    output_path = input_path + '.enc'
    with open(output_path, 'wb') as f:
        f.write(salt + iv + ciphertext)

    messagebox.showinfo("Success", f"File encrypted: {output_path}")

def decrypt_file(input_path, password):
    with open(input_path, 'rb') as f:
        data = f.read()

    salt = data[:SALT_SIZE]
    iv = data[SALT_SIZE:SALT_SIZE + BLOCK_SIZE]
    ciphertext = data[SALT_SIZE + BLOCK_SIZE:]

    key = derive_key(password, salt)
    cipher = Cipher(algorithms.AES(key), modes.CBC(iv), backend=default_backend())
    decryptor = cipher.decryptor()
    plaintext_padded = decryptor.update(ciphertext) + decryptor.finalize()

    padding_length = plaintext_padded[-1]
    plaintext = plaintext_padded[:-padding_length]

    output_path = input_path.replace('.enc', '.dec')
    with open(output_path, 'wb') as f:
        f.write(plaintext)

    messagebox.showinfo("Success", f"File decrypted: {output_path}")

def browse_file():
    file_path.set(filedialog.askopenfilename())

def run_encrypt():
    if not file_path.get() or not password.get():
        messagebox.showerror("Error", "Please select a file and enter a password.")
        return
    encrypt_file(file_path.get(), password.get())

def run_decrypt():
    if not file_path.get() or not password.get():
        messagebox.showerror("Error", "Please select a file and enter a password.")
        return
    decrypt_file(file_path.get(), password.get())

# GUI Setup
root = tk.Tk()
root.title("SecureVault File Encrypter")

file_path = tk.StringVar()
password = tk.StringVar()
m=tk.Menu(root)
root.config(menu=m)
file_menu=tk.Menu(m)
m.add_cascade(label="Exit",command=root.quit)
tk.Label(root, text="File Path:").grid(row=0, column=0, padx=10, pady=10)
tk.Entry(root, textvariable=file_path, width=40).grid(row=0, column=1)
tk.Button(root, text="Browse", command=browse_file).grid(row=0, column=2)

tk.Label(root, text="Password:").grid(row=1, column=0, padx=10, pady=10)
tk.Entry(root, textvariable=password, show="*", width=40).grid(row=1, column=1)

tk.Button(root, text="Encrypt", command=run_encrypt, width=20).grid(row=2, column=0, pady=20)
tk.Button(root, text="Decrypt", command=run_decrypt, width=20).grid(row=2, column=1)

root.mainloop()
